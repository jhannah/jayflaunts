And then let's merge our 2 SQLite databases (carefully because of the overlap)

sqlite3 downloads.db <<'SQL'

ATTACH 'downloads.db.201710-202105' AS old;
ATTACH 'downloads.db.202101-202604' AS new;

CREATE TABLE downloads (
    id          INTEGER PRIMARY KEY,
    timestamp   TEXT NOT NULL,
    remote_ip   TEXT, operation TEXT, key TEXT,
    http_status INTEGER, bytes_sent INTEGER, object_size INTEGER,
    referer TEXT, user_agent TEXT, log_file TEXT
);
CREATE INDEX idx_timestamp ON downloads(timestamp);
CREATE INDEX idx_key       ON downloads(key);

CREATE TABLE processed_months (
    month        TEXT PRIMARY KEY,
    processed_at TEXT NOT NULL
);

-- db1: everything strictly before the overlap
INSERT INTO downloads (timestamp, remote_ip, operation, key, http_status, bytes_sent, object_size, referer, user_agent, log_file)
  SELECT timestamp, remote_ip, operation, key, http_status, bytes_sent, object_size, referer, user_agent, log_file
  FROM old.downloads
  WHERE strftime('%Y-%m', timestamp) < '2021-01';

-- db2: all rows (owns the overlap and everything after)
INSERT INTO downloads (timestamp, remote_ip, operation, key, http_status, bytes_sent, object_size, referer, user_agent, log_file)
  SELECT timestamp, remote_ip, operation, key, http_status, bytes_sent, object_size, referer, user_agent, log_file
  FROM new.downloads;

-- same pattern for processed_months
INSERT INTO processed_months SELECT * FROM old.processed_months WHERE month < '2021-01';
INSERT INTO processed_months SELECT * FROM new.processed_months;
